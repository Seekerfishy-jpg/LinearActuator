import asyncio

# Network Configuration
LISTEN_HOST = "0.0.0.0"
LISTEN_PORT = 8800

# Actuator Config
TRAVEL_SPEED = 10.0  # mm/s

#Inputs
inputs = [
    ["1", 50],
    ["2", 50],
    ["3", 50],
    ["4", 50],
    ["5", 50],
    ["6", 50],
]

# Pre-converted byte objects for direct socket writing
relays = {
    "1": {

            "OFF": bytes.fromhex("01 05 00 00 00 00 CD CA"),  # Relay 1

            "OFF": bytes.fromhex("01 05 00 0B 00 00 79 CE"),  # Relay 12
        },
    },
}

# Declared without initialization
stop_server_event = None


async def handle_board(
    reader: asyncio.StreamReader, writer: asyncio.StreamWriter
):
    client_addr = writer.get_extra_info("peername")
    print(f"[SERVER] Board connected from {client_addr}")
    
    try:
        for relay_needed, distance_needed in inputs:
            if distance_needed == 0:
                print(f"[SERVER] Actuator {relay_needed}: Distance is 0, skipping.")
                continue

            direction = "EXTEND" if distance_needed > 0 else "RETRACT"
            duration = abs(distance_needed) / TRAVEL_SPEED

            print(
                f"[SERVER] Actuator {relay_needed}: Moving {direction} "
                f"for {distance_needed}mm ({duration:.2f}s)"
            )

            # Turn relay ON
            writer.write(relays[relay_needed][direction]["ON"])
            await writer.drain()

            # Wait for travel time
            await asyncio.sleep(duration)

            # Turn relay OFF
            writer.write(relays[relay_needed][direction]["OFF"])
            await writer.drain()

            # Small inter-command delay to allow physical relay settling
            await asyncio.sleep(0.1)
    except Exception as e:
        print()

    finally:
            # Safety fallback: Send OFF command and close connection
            writer.write
            await writer.drain()
            writer.close()
            await writer.wait_closed()
            print("[SERVER] Operation completed. Closing connection.")
            # Unblock main() so the program exits cleanly
            stop_server_event.set()


async def main():
    global stop_server_event
    # Initialize Event inside the running loop
    stop_server_event = asyncio.Event()

    server = await asyncio.start_server(handle_board, LISTEN_HOST, LISTEN_PORT)
    print(
        f"[SERVER] Listening on port {LISTEN_PORT}... Waiting for 1 connection."
    )

    async with server:
        # Wait until handle_board sets the stop signal
        await stop_server_event.wait()

    print("[SERVER] Execution finished. Server shut down.")


if __name__ == "__main__":
    asyncio.run(main())
