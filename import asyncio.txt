def init_joysticks():
    pygame.init()
    pygame.display.set_caption("ACTUATOR WINDOW - KEEP FOCUS HERE")
    pygame.display.set_mode((300, 100))

    pygame.joystick.init()
    count = pygame.joystick.get_count()
    if count < 2:
        logging.error("Less than 2 USB controllers detected!")
        sys.exit(1)


    instance_to_index = {}
    for i in range(count):
        js = pygame.joystick.Joystick(i)
        js.init()
        instance_to_index[js.get_instance_id()] = i
        #print(f"[INIT] Controller {i}: {js.get_name()} (Instance ID: {js.get_instance_id()})")
        print(instance_to_index)
    return instance_to_index
init_joysticks()

async def send_cmd(cmd: bytes):
    global active_writer, active_reader
    if active_writer is None or active_writer.is_closing():
        print("[NETWORK] Relay board is not connected!")
        return

    try:
        active_writer.write(cmd)
        await active_writer.drain()
        await asyncio.wait_for(active_reader.read(1024), timeout=0.2)
    except asyncio.TimeoutError:
        pass
    except Exception as e:
        print(f"[NETWORK ERROR] Socket exception: {e}")

async def emergency_shutdown():
    TURN_ALL_OFF = bytes.fromhex("01 0F 00 00 00 10 02 00 00 E2 20")
    global e_stop_triggered
    e_stop_triggered = True
    print("\n[CRITICAL] !!! EMERGENCY STOP TRIGGERED !!! TURNING OFF ALL RELAYS.\n")
    await send_cmd(TURN_ALL_OFF)

async def run_auto_sequence(instance_to_index):
    global current_mode, e_stop_triggered

    print("\n==========================================")
    print("      AUTOMATED SEQUENCE MODE (AUTO)      ")
    print("==========================================")

    # --- STEP 1: Collect inputs upfront (Type 'q' to quit) ---
    auto_inputs = []
    print("\n[STEP 1] Enter target distances for each relay.")
    print("Type distance in mm (e.g., 50 for EXTEND, -30 for RETRACT, 0 to SKIP).")
    print("Type 'q' at any prompt to exit back to MANUAL mode.\n")

    for i in range(1, 7):
        if current_mode != "AUTO" or e_stop_triggered:
            print("[AUTO] Sequence setup cancelled.")
            return

        pygame.event.pump()  # Keep Pygame window responsive while typing
        val_raw = input(f"Relay {i} Distance (mm) [or 'q']: ").strip()

        # Keyboard 'q' exit
        if val_raw.lower() == 'q':
            current_mode = "MANUAL"
            print("\n[SYSTEM] Returning to MANUAL mode...")
            return

        try:
            dist = float(val_raw)
        except ValueError:
            print("  -> Invalid input! Defaulting to 0 mm (SKIP).")
            dist = 0.0

        auto_inputs.append((str(i), dist))

    # --- STEP 2: Execute relay movements sequentially ---
    print("\n[STEP 2] Starting physical movement sequence...")
    print("Controller overrides: Button 13 -> MANUAL MODE | Button 15 -> EMERGENCY STOP\n")

    for relay_needed, distance_needed in auto_inputs:
        if current_mode != "AUTO" or e_stop_triggered:
            print("[AUTO] Sequence interrupted by user!")
            break

        if distance_needed == 0:
            print(f"[AUTO] Relay {relay_needed}: 0 mm -> Skipping.")
            continue

        direction = "EXTEND" if distance_needed > 0 else "RETRACT"
        duration = abs(distance_needed) / TRAVEL_SPEED

        print(f"[AUTO] Relay {relay_needed}: Moving {direction} ({abs(distance_needed)} mm @ {duration:.2f}s)...")
        await send_cmd(relays[relay_needed][direction]["ON"])

        start_time = asyncio.get_running_loop().time()
        while asyncio.get_running_loop().time() - start_time < duration:
            for event in pygame.event.get():
                if event.type == pygame.QUIT:
                    sys.exit(0)
                elif event.type == pygame.JOYBUTTONDOWN:
                    idx = instance_to_index.get(event.instance_id)
                    if idx is not None and event.button in button_mapping[idx]:
                        btn = button_mapping[idx][event.button]
                        if btn == 13:
                            print("\n[SYSTEM] Button 13 pressed -> Aborting AUTO mode!")
                            current_mode = "MANUAL"
                        elif btn == 15:
                            asyncio.create_task(emergency_shutdown())

            if current_mode != "AUTO" or e_stop_triggered:
                await send_cmd(relays[relay_needed][direction]["OFF"])
                print(f"[AUTO] Relay {relay_needed} stopped early.")
                return

            await asyncio.sleep(0.01)

        await send_cmd(relays[relay_needed][direction]["OFF"])
        await asyncio.sleep(0.1)

    # --- STEP 3: Finished sequence -> Prompt for next set ---
    if current_mode == "AUTO" and not e_stop_triggered:
        print("\n[STEP 3] Sequence completed successfully!")
        pygame.event.pump()
        choice = input("\nRun another auto sequence? [Y/n or 'q' for manual]: ").strip().lower()
        if choice in ('n', 'q'):
            current_mode = "MANUAL"
            print("[SYSTEM] Switching to MANUAL mode...\n")

async def poll_joysticks():
    while True:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                sys.exit("EXIT")
            if event.type == pygame.JOYBUTTONDOWN or pygame.JOYBUTTONUP:
                mapped_index = instance_to_index.get(event.instance_id)
                if mapped_idx is not None and event.button in button_mapping[mapped_idx]:
                    btn = button_mapping[mapped_idx][event.button]
            if event.type == pygame.JOYBUTTONDOWN:
                if btn == 1:
                    current_mode = "MANUAL"
                    e_stop_triggered = False
                elif btn == 2:
                    current_mode = "AUTO"
                    e_stop_triggered = False
                elif btn == 3:
                    asyncio.create_task(emergency_shutdown)
            if current_mode == "MANUAL" and e_stop_triggered == False and btn > 4:
                if event.type == pygame.JOYBUTTONDOWN:
                    power = "ON"
                elif event.type == pygame.JOYBUTTONUP:
                    power = "OFF"
                else:
                    print("Check USB Connections again")
                if btn % 2 == 0:
                    direction = "EXTEND"
                    relay_needed = (btn // 2) - 2
                else:
                    direction = "RETRACT"
                    relay_needed = (btn // 2) - 1
                print(f"[MANUAL] Button: {btn} | Relay: {relay_needed} | Dir: {direction} | Power: {power}")
                cmd = relays[relay_needed][direction][power]
                await send_cmd(cmd)
        if current_mode == "AUTO" and e_stop_triggered == False:
            await run_auto_sequence(instance_to_index)
        await asyncio.sleep(1)
        print("Switching to audo mode")

async def handle_board(reader: asyncio.StreamReader, writer: asyncio.StreamWriter):
    global active_writer, active_reader
    active_writer = writer
    active_reader = reader
    print(f"\n*** ETHERNET BOARD CONNECTED from {writer.get_extra_info('peername')} ***\n")
    try:
        while not writer.is_closing():
            data = await reader.read(100)
            if not data:
                break
    except Exception:
        pass
    finally:
        print("\n*** BOARD DISCONNECTED ***\n")
        active_writer = None
        active_reader = None

async def main():
    instance_to_index = init_joysticks()

    # Start TCP server in background
    server = await asyncio.start_server(handle_board, LISTEN_HOST, LISTEN_PORT)
    print(f"[SERVER] Listening on port {LISTEN_PORT}... Waiting for Relay Board connection.")

    # Run joystick polling immediately
    asyncio.create_task(poll_joysticks(instance_to_index))

    async with server:
        await server.serve_forever()


if __name__ == "__main__":
    asyncio.run(main())
