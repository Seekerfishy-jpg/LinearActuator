import asyncio

HOST = "0.0.0.0"
PORT = 8800

# Modbus command: Relay 1 RETRACT ON
RELAY_1_ON = bytes.fromhex("01 05 00 00 FF 00 8C 3A")
RELAY_1_OFF = bytes.fromhex("01 05 00 00 00 00 CD CA")

async def handle_client(reader, writer):
    print(f"\n[SUCCESS] Relay board connected from: {writer.get_extra_info('peername')}")
    
    print("Testing Relay 1 ON in 2 seconds...")
    await asyncio.sleep(2)
    writer.write(RELAY_1_ON)
    await writer.drain()
    print("--> Sent Relay 1 ON")

    await asyncio.sleep(2)
    writer.write(RELAY_1_OFF)
    await writer.drain()
    print("--> Sent Relay 1 OFF")
    
    writer.close()
    await writer.wait_closed()
    print("Test complete.")

async def main():
    server = await asyncio.start_server(handle_client, HOST, PORT)
    print(f"Server listening on port {PORT}... Waiting for board to connect.")
    async with server:
        await server.serve_forever()

if __name__ == "__main__":
    asyncio.run(main())
