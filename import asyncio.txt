import pygame
import sys

pygame.init()
pygame.display.set_caption("USB Encoder Test - KEEP THIS WINDOW FOCUSED")
screen = pygame.display.set_mode((400, 150))
pygame.joystick.init()

count = pygame.joystick.get_count()
print(f"\nDetected {count} USB controller(s).")

if count == 0:
    print("ERROR: No USB controllers detected by Pygame!")
    sys.exit()

for i in range(count):
    js = pygame.joystick.Joystick(i)
    js.init()
    print(f"Controller {i}: {js.get_name()} | Instance ID: {js.get_instance_id()}")

print("\n--- Move/Toggle your Rocker Switch now ---")

clock = pygame.time.Clock()
while True:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            pygame.quit()
            sys.exit()

        if event.type == pygame.JOYBUTTONDOWN:
            print(f"[BUTTON DOWN] Controller InstID: {event.instance_id} | Button Index: {event.button}")
        elif event.type == pygame.JOYBUTTONUP:
            print(f"[BUTTON UP]   Controller InstID: {event.instance_id} | Button Index: {event.button}")

    # Draw simple text to ensure window updates
    screen.fill((30, 30, 30))
    pygame.display.flip()
    clock.tick(30)
