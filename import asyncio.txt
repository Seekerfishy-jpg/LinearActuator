import asyncio
import logging
import math
import os
import sys

# Lets pygame receive controller buttons even when its window isn't focused.
# Must be set before pygame is initialised.
os.environ.setdefault("SDL_JOYSTICK_ALLOW_BACKGROUND_EVENTS", "1")
import pygame

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s")

# ---------------------------------------------------------------- config
LISTEN_HOST = "0.0.0.0"
LISTEN_PORT = 8800
TRAVEL_SPEED = 10.0     # mm/s
MAX_STROKE_MM = 100.0   # TODO: set to your actuator's real stroke length
NUM_ACTUATORS = 6

# ---------------------------------------------------------------- state
current_mode = "MANUAL"
e_stop_triggered = False

active_writer = None
send_lock = None        # created in main()
background_tasks = set()
joysticks = []          # keep references or pygame stops delivering events


def spawn(coro):
    """create_task, but keep a reference so the task can't be garbage collected."""
    task = asyncio.create_task(coro)
    background_tasks.add(task)
    task.add_done_callback(background_tasks.discard)
    return task


# ---------------------------------------------------------------- Modbus RTU
def crc16(data: bytes) -> bytes:
    crc = 0xFFFF
    for b in data:
        crc ^= b
        for _ in range(8):
            crc = (crc >> 1) ^ 0xA001 if crc & 1 else crc >> 1
    return crc.to_bytes(2, "little")


def coil_cmd(coil: int, on: bool) -> bytes:
    frame = bytes([0x01, 0x05, 0x00, coil, 0xFF if on else 0x00, 0x00])
    return frame + crc16(frame)


# Relay n: RETRACT = coil 2*(n-1), EXTEND = coil 2*(n-1)+1
relays = {
    str(n): {
        "RETRACT": {"ON": coil_cmd(2 * (n - 1), True), "OFF": coil_cmd(2 * (n - 1), False)},
        "EXTEND": {"ON": coil_cmd(2 * (n - 1) + 1, True), "OFF": coil_cmd(2 * (n - 1) + 1, False)},
    }
    for n in range(1, NUM_ACTUATORS + 1)
}

_all_off_frame = bytes.fromhex("01 0F 00 00 00 10 02 00 00")  # write 16 coils = 0
TURN_ALL_OFF = _all_off_frame + crc16(_all_off_frame)

# controller 0 -> logical buttons 1-8, controller 1 -> logical buttons 9-16
button_mapping = {c: {b: c * 8 + b + 1 for b in range(8)} for c in range(2)}


# ---------------------------------------------------------------- controllers
def init_joysticks():
    pygame.init()
    pygame.display.set_caption("ACTUATOR WINDOW")
    pygame.display.set_mode((300, 100))

    pygame.joystick.init()
    count = pygame.joystick.get_count()
    if count < 2:
        logging.error("Less than 2 USB controllers detected!")
        sys.exit(1)

    instance_to_index = {}
    for i in range(count):
        js = pygame.joystick.Joystick(i)
        joysticks.append(js)
        instance_to_index[js.get_instance_id()] = i
        logging.info("Controller %d: %s (instance %d)", i, js.get_name(), js.get_instance_id())
    return instance_to_index


def logical_button(instance_to_index, event):
    idx = instance_to_index.get(event.instance_id)
    return button_mapping.get(idx, {}).get(event.button)


# ---------------------------------------------------------------- network
async def send_cmd(cmd: bytes) -> bool:
    if active_writer is None or active_writer.is_closing():
        logging.warning("[NETWORK] Relay board is not connected!")
        return False
    try:
        async with send_lock:
            active_writer.write(cmd)
            await active_writer.drain()
        return True
    except Exception as e:
        logging.error("[NETWORK ERROR] %s", e)
        return False


async def emergency_shutdown():
    global e_stop_triggered
    e_stop_triggered = True
    logging.critical("!!! EMERGENCY STOP !!! TURNING OFF ALL RELAYS")
    await send_cmd(TURN_ALL_OFF)


async def handle_board(reader: asyncio.StreamReader, writer: asyncio.StreamWriter):
    """Only this function reads from the socket (a StreamReader allows one reader)."""
    global active_writer
    active_writer = writer
    logging.info("*** BOARD CONNECTED from %s ***", writer.get_extra_info("peername"))
    try:
        while True:
            data = await reader.read(100)
            if not data:
                break
            logging.debug("Board reply: %s", data.hex(" "))
    except Exception:
        pass
    finally:
        logging.warning("*** BOARD DISCONNECTED ***")
        if active_writer is writer:
            active_writer = None


# ---------------------------------------------------------------- auto mode
def handle_auto_events(instance_to_index):
    global current_mode
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            raise SystemExit(0)
        if event.type != pygame.JOYBUTTONDOWN:
            continue
        btn = logical_button(instance_to_index, event)
        if btn in (1, 13):
            logging.info("Button %s -> aborting AUTO mode", btn)
            current_mode = "MANUAL"
        elif btn in (3, 15):  # 3 is the e-stop in manual mode, so accept it here too
            spawn(emergency_shutdown())


async def ask(prompt: str) -> str:
    # input() blocks; run it in a thread so the event loop (and the board
    # connection) keep running while the user types.
    return (await asyncio.to_thread(input, prompt)).strip()


async def run_auto_sequence(instance_to_index):
    global current_mode

    print("\n==========================================")
    print("      AUTOMATED SEQUENCE MODE (AUTO)      ")
    print("==========================================")
    print(f"\nEnter target distance in mm for each relay (max +/-{MAX_STROKE_MM:g}).")
    print("Positive = EXTEND, negative = RETRACT, 0 = SKIP, 'q' = back to MANUAL.\n")

    auto_inputs = []
    for i in range(1, NUM_ACTUATORS + 1):
        while True:
            if current_mode != "AUTO" or e_stop_triggered:
                print("[AUTO] Sequence setup cancelled.")
                return
            raw = await ask(f"Relay {i} distance (mm) [or 'q']: ")
            if raw.lower() == "q":
                current_mode = "MANUAL"
                print("\n[SYSTEM] Returning to MANUAL mode...")
                return
            try:
                dist = float(raw)
            except ValueError:
                print("  -> Not a number, try again.")
                continue
            if not math.isfinite(dist) or abs(dist) > MAX_STROKE_MM:
                print(f"  -> Must be between -{MAX_STROKE_MM:g} and {MAX_STROKE_MM:g} mm.")
                continue
            auto_inputs.append((str(i), dist))
            break

    print("\nStarting movement. Button 1 or 13 -> MANUAL | Button 3 or 15 -> E-STOP\n")
    loop = asyncio.get_running_loop()

    for relay_needed, distance in auto_inputs:
        if current_mode != "AUTO" or e_stop_triggered:
            print("[AUTO] Sequence interrupted!")
            return
        if distance == 0:
            print(f"[AUTO] Relay {relay_needed}: 0 mm -> skipping.")
            continue

        direction = "EXTEND" if distance > 0 else "RETRACT"
        duration = abs(distance) / TRAVEL_SPEED
        print(f"[AUTO] Relay {relay_needed}: {direction} {abs(distance):g} mm ({duration:.2f}s)")

        try:
            start = loop.time()  # start the clock when ON is sent
            await send_cmd(relays[relay_needed][direction]["ON"])
            while loop.time() - start < duration:
                handle_auto_events(instance_to_index)
                if current_mode != "AUTO" or e_stop_triggered:
                    print(f"[AUTO] Relay {relay_needed} stopped early.")
                    return
                await asyncio.sleep(0.005)
        finally:
            # Always runs: normal finish, early stop, or crash
            await send_cmd(relays[relay_needed][direction]["OFF"])
        await asyncio.sleep(0.1)

    print("\nSequence completed.")
    choice = (await ask("Run another auto sequence? [Y/n or 'q' for manual]: ")).lower()
    if choice in ("n", "q"):
        current_mode = "MANUAL"
        print("[SYSTEM] Switching to MANUAL mode...\n")


# ---------------------------------------------------------------- main loop
async def poll_joysticks(instance_to_index):
    global current_mode, e_stop_triggered

    while True:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                await send_cmd(TURN_ALL_OFF)
                raise SystemExit(0)
            if event.type not in (pygame.JOYBUTTONDOWN, pygame.JOYBUTTONUP):
                continue

            btn = logical_button(instance_to_index, event)
            if btn is None:
                continue
            pressed = event.type == pygame.JOYBUTTONDOWN

            # Mode / e-stop buttons
            if btn in (1, 2, 3):
                if not pressed:
                    continue
                if btn == 3:
                    spawn(emergency_shutdown())
                else:
                    await send_cmd(TURN_ALL_OFF)  # nothing stays on across a mode change
                    e_stop_triggered = False
                    new_mode = "MANUAL" if btn == 1 else "AUTO"
                    if new_mode != current_mode:
                        logging.info("Switching to %s mode", new_mode)
                    current_mode = new_mode
                continue

            # Manual jog buttons: 5..16 -> relays 1..6, odd = RETRACT, even = EXTEND
            if btn >= 5:
                relay_needed = str((btn - 5) // 2 + 1)
                direction = "RETRACT" if btn % 2 == 1 else "EXTEND"
                opposite = "EXTEND" if direction == "RETRACT" else "RETRACT"

                if pressed:
                    if current_mode != "MANUAL" or e_stop_triggered:
                        continue  # ignore new motion, but never ignore an OFF
                    logging.info("[MANUAL] btn %d | relay %s | %s | ON", btn, relay_needed, direction)
                    await send_cmd(relays[relay_needed][opposite]["OFF"])  # interlock
                    await send_cmd(relays[relay_needed][direction]["ON"])
                else:
                    logging.info("[MANUAL] btn %d | relay %s | %s | OFF", btn, relay_needed, direction)
                    await send_cmd(relays[relay_needed][direction]["OFF"])

        if current_mode == "AUTO" and not e_stop_triggered:
            await run_auto_sequence(instance_to_index)

        await asyncio.sleep(0.01)


async def main():
    global send_lock
    send_lock = asyncio.Lock()

    instance_to_index = init_joysticks()

    server = await asyncio.start_server(handle_board, LISTEN_HOST, LISTEN_PORT)
    logging.info("[SERVER] Listening on port %d... waiting for relay board.", LISTEN_PORT)

    poll_task = asyncio.create_task(poll_joysticks(instance_to_index))
    async with server:
        # gather so that a crash in poll_joysticks is raised instead of vanishing
        await asyncio.gather(server.serve_forever(), poll_task)


if __name__ == "__main__":
    try:
        asyncio.run(main())
    except (KeyboardInterrupt, SystemExit):
        pass
