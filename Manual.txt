import sys
import pygame

def test_controllers():
    pygame.init()
    pygame.display.set_caption("BUTTON MAP TESTER - KEEP FOCUS HERE")
    pygame.display.set_mode((300, 100))

    pygame.joystick.init()
    count = pygame.joystick.get_count()
    print(f"\n[INIT] Detected {count} joystick(s).")

    if count == 0:
        print("[ERROR] No controllers found! Please check your USB connections.")
        sys.exit(1)

    instance_to_index = {}
    for i in range(count):
        js = pygame.joystick.Joystick(i)
        js.init()
        instance_id = js.get_instance_id()
        instance_to_index[instance_id] = i
        print(f"  -> Controller Index {i}: {js.get_name()} (Instance ID: {instance_id})")

    print("\n==================================================")
    print(" PRESS ANY BUTTON / FLIP ANY SWITCH ON YOUR CONTROLLERS")
    print(" Press Ctrl+C or close this window to exit.")
    print("==================================================\n")

    while True:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                pygame.quit()
                sys.exit(0)

            if event.type in (pygame.JOYBUTTONDOWN, pygame.JOYBUTTONUP):
                inst_id = event.instance_id
                ctrl_idx = instance_to_index.get(inst_id, "Unknown")
                raw_btn = event.button
                action = "DOWN (Pressed)" if event.type == pygame.JOYBUTTONDOWN else "UP (Released)"
                
                print(f"Controller Index: [{ctrl_idx}] (Instance ID: {inst_id}) | Button ID: {raw_btn:2d} | State: {action}")

if __name__ == "__main__":
    test_controllers()
